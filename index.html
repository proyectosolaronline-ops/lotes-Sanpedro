// ========== SISTEMA DE ACCESO POR CÓDIGO ========== const CODIGOS_TEMP = {}; function solicitarCodigoAcceso(correo) { correo = String(correo || '').trim().toLowerCase(); if (!correo || !correo.includes('@')) throw new Error('Correo inválido.'); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(NOMBRE_HOJA_USUARIOS); const data = sheet.getDataRange().getValues(); const headers = data.shift().map(h => String(h || '').trim().toLowerCase()); const idxCorreo = headers.indexOf('correo'); const idxEstado = headers.indexOf('estado'); const idxNombre = headers.indexOf('nombre'); const fila = data.find(row => String(row[idxCorreo] || '').trim().toLowerCase() === correo); if (!fila) throw new Error('Correo no registrado.'); if (String(fila[idxEstado] || '').trim().toLowerCase() !== 'activo') throw new Error('Cuenta inactiva.'); const codigo = String(Math.floor(100000 + Math.random() * 900000)); CODIGOS_TEMP[correo] = { codigo, expira: Date.now() + 600000, intentos: 0 }; const nombre = fila[idxNombre] || 'Usuario'; GmailApp.sendEmail(correo, 'Código de acceso: ' + codigo, 'Hola ' + nombre + ',\n\nTu código de acceso es: ' + codigo + '\n\nVálido por 10 minutos.'); return { success: true, mensaje: 'Código enviado' }; } function verificarCodigoAcceso(correo, codigo) { correo = String(correo || '').trim().toLowerCase(); codigo = String(codigo || '').trim(); const reg = CODIGOS_TEMP[correo]; if (!reg) throw new Error('Solicita un código primero.'); if (Date.now() > reg.expira) { delete CODIGOS_TEMP[correo]; throw new Error('Código expirado.'); } if (reg.intentos >= 3) { delete CODIGOS_TEMP[correo]; throw new Error('Demasiados intentos.'); } reg.intentos++; if (reg.codigo !== codigo) throw new Error('Código incorrecto. Intentos: ' + (3 - reg.intentos)); delete CODIGOS_TEMP[correo]; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(NOMBRE_HOJA_USUARIOS); const data = sheet.getDataRange().getValues(); const headers = data.shift().map(h => String(h || '').trim().toLowerCase()); const fila = data.find(row => String(row[headers.indexOf('correo')] || '').trim().toLowerCase() === correo); return { autorizado: true, nombre: fila[headers.indexOf('nombre')] || correo }; }
